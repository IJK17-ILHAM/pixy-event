function renderAdminTable() { const tbody = document.getElementById('participantTableBody'); const filter = document.getElementById('filterAcaraTable').value; const filtered = filter === "ALL" ? participants : participants.filter(p => { const valAcara = getVal(p, ['Pilihan Acara', 'acara', 'Acara', 'Pilihan']); return valAcara.toLowerCase().includes(filter.toLowerCase()) || filter.toLowerCase().includes(valAcara.toLowerCase()); }); if (filtered.length === 0) { tbody.innerHTML = `Belum ada pendaftar di acara ini.`; return; } tbody.innerHTML = filtered.map((p, index) => { const nama = getVal(p, ['Nama Lengkap', 'nama', 'Nama']) || '-'; const wa = getVal(p, ['WhatsApp', 'whatsapp', 'Whatspp', 'Kontak', 'No WhatsApp']) || '-'; const email = getVal(p, ['Email', 'email']) || '-'; const acara = getVal(p, ['Pilihan Acara', 'acara', 'Acara', 'Pilihan']) || '-'; const kategori = getVal(p, ['Kategori', 'kategori']) || '-'; let bukti = getVal(p, ['Bukti Pembayaran', 'bukti', 'Bukti', 'Bukti Transfer']); const statusBayar = getVal(p, ['Status Bayar', 'statusBayar', 'Status']) || 'Menunggu'; const kehadiran = getVal(p, ['Kehadiran', 'kehadiran']) || 'Belum Hadir'; // Deteksi apakah ada link Google Drive yang valid const isUrlValid = bukti.startsWith('http'); return `

${nama}

${kategori}

${wa}

${email}

${acara} ${isUrlValid ? ` Buka Bukti ` : `Tidak ada bukti` } ${statusBayar} ${kehadiran} `; }).join(''); }
